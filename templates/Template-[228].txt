'use client';
import { useEffect, useRef } from 'react';
const EmojiRain = ({ set = ['⚡', '✦', '●', '▲', '■'], n = 34 }) => {
  const ref = useRef(null);
  useEffect(() => {
    const c = ref.current, x = c.getContext('2d');
    const P = Array.from({ length: n }, () => ({
      e: set[Math.floor(Math.random() * set.length)],
      x: Math.random() * c.width, y: Math.random() * -c.height,
      v: 0.8 + Math.random() * 2, s: 12 + Math.random() * 12, r: (Math.random() - 0.5) * 0.1, a: 0 }));
    x.font = '16px serif'; x.textBaseline = 'top';
    let raf = 0;
    const tick = () => { x.clearRect(0, 0, c.width, c.height);
      P.forEach(p => { p.y += p.v; p.a += p.r;
        if (p.y > c.height + 20) { p.y = -20; p.x = Math.random() * c.width; }
        x.save(); x.translate(p.x, p.y); x.rotate(p.a); x.font = `${p.s}px serif`;
        x.fillText(p.e, 0, 0); x.restore(); });
      raf = requestAnimationFrame(tick); };
    raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, []);
  return <canvas ref={ref} width={320} height={180} className="fx-emojrain" />;
};
export default EmojiRain;

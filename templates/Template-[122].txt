'use client';
import { useEffect, useRef } from 'react';
const BubbleRise = ({ bubbles = 26 }) => {
  const ref = useRef(null);
  useEffect(() => {
    const c = ref.current, x = c.getContext('2d');
    const spawn = () => ({ x: Math.random() * c.width, y: c.height + 10,
      v: 0.4 + Math.random(), r: 2 + Math.random() * 6, w: Math.random() * 6.28 });
    const B = Array.from({ length: bubbles }, spawn);
    let raf = 0;
    const tick = () => {
      const g = x.createLinearGradient(0, 0, 0, c.height);
      g.addColorStop(0, '#06283d'); g.addColorStop(1, '#03121e');
      x.fillStyle = g; x.fillRect(0, 0, c.width, c.height);
      x.fillStyle = 'rgba(120,200,240,0.25)'; x.fillRect(0, 0, c.width, 3);
      B.forEach((b, i) => { b.y -= b.v; b.w += 0.02; b.x += Math.sin(b.w) * 0.5;
        if (b.y < 6) B[i] = spawn();
        x.strokeStyle = 'rgba(160,220,250,0.7)'; x.lineWidth = 1;
        x.beginPath(); x.arc(b.x, b.y, b.r, 0, 6.29); x.stroke();
        x.fillStyle = 'rgba(255,255,255,0.5)';
        x.beginPath(); x.arc(b.x - b.r * 0.3, b.y - b.r * 0.3, b.r * 0.25, 0, 6.29); x.fill(); });
      raf = requestAnimationFrame(tick); };
    raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, [bubbles]);
  return <canvas ref={ref} width={320} height={180} className="fx-bubble" />;
};
export default BubbleRise;

'use client';
import { useEffect, useRef } from 'react';
const MoneyRain = ({ n = 22 }) => {
  const ref = useRef(null);
  useEffect(() => {
    const c = ref.current, x = c.getContext('2d');
    const P = Array.from({ length: n }, () => ({
      x: Math.random() * c.width, y: Math.random() * -c.height,
      v: 0.7 + Math.random() * 1.4, w: Math.random() * 6.28, s: 0.7 + Math.random() * 0.7 }));
    let raf = 0;
    const tick = () => { x.fillStyle = '#07130b'; x.fillRect(0, 0, c.width, c.height);
      P.forEach(p => { p.y += p.v; p.w += 0.03;
        if (p.y > c.height + 16) { p.y = -16; p.x = Math.random() * c.width; }
        x.save(); x.translate(p.x + Math.sin(p.w) * 12, p.y); x.rotate(Math.sin(p.w) * 0.5); x.scale(p.s, p.s);
        x.fillStyle = '#0f7a3d'; x.fillRect(-14, -8, 28, 16);
        x.strokeStyle = '#c9f2d4'; x.strokeRect(-14, -8, 28, 16);
        x.fillStyle = '#e9fff0'; x.font = 'bold 10px monospace'; x.fillText('$', -3, 4);
        x.restore(); });
      raf = requestAnimationFrame(tick); };
    raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, [n]);
  return <canvas ref={ref} width={320} height={180} className="fx-money" />;
};
export default MoneyRain;

'use client';

import { useEffect, useState } from 'react';
import './Equalizer.css';

const BARS = 24;

const Equalizer = ({ color = '#00F0FF', idle = true }) => {
  const [levels, setLevels] = useState(Array(BARS).fill(0.2));

  useEffect(() => {
    if (!idle) return;
    const iv = setInterval(() => {
      setLevels(prev =>
        prev.map(v => {
          const drift = (Math.random() - 0.52) * 0.35;
          return Math.min(1, Math.max(0.06, v + drift));
        })
      );
    }, 120);
    return () => clearInterval(iv);
  }, [idle]);

  return (
    <div className="eq">
      {levels.map((v, i) => (
        <div
          key={i}
          className="eq-bar"
          style={{ height: `${v * 100}%`, background: color, animationDelay: `${i * 40}ms` }}
        />
      ))}
    </div>
  );
};

export default Equalizer;

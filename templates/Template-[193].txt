'use client';
import { useEffect, useState } from 'react';
import './VoiceWave.css';
const VoiceWave = ({ bars = 18 }) => {
  const [live, setLive] = useState(true);
  const [tick, setTick] = useState(0);
  useEffect(() => { const iv = setInterval(() => setTick(t => t + 1), 140);
    return () => clearInterval(iv); }, []);
  return <button onClick={() => setLive(!live)} className={`voicewave${live ? '' : ' paused'}`}
    aria-label="toggle mic"><span className="voicewave-mic">🎙</span>
    <span className="voicewave-bars">{Array.from({ length: bars }, (_, i) => (
      <i key={`${tick}-${i}`} style={{ transform: `scaleY(${live ? 0.15 + Math.random() * 0.85 : 0.12})` }} />))}</span></button>;
};
export default VoiceWave;

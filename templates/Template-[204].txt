'use client';
import { useEffect, useState } from 'react';
import './CalendarFlip.css';
const CalendarFlip = () => {
  const [day, setDay] = useState(6);
  const [flip, setFlip] = useState(0);
  useEffect(() => { const iv = setInterval(() => {
      setFlip(f => f + 1); setDay(d => (d % 30) + 1);
    }, 3000); return () => clearInterval(iv); }, []);
  return <div className="calflip"><div className="calflip-top">OCT</div>
    <div className="calflip-day" key={flip}>{day}</div></div>;
};
export default CalendarFlip;

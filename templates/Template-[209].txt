'use client';
import { useEffect, useState } from 'react';
import './TabsSlide.css';
const TabsSlide = ({ tabs = ['voice', 'mcp', 'rag'] }) => {
  const [i, setI] = useState(0);
  useEffect(() => { const iv = setInterval(() => setI(v => (v + 1) % tabs.length), 1800);
    return () => clearInterval(iv); }, [tabs.length]);
  return <div className="tabslide">
    <div className="tabslide-ind" style={{ transform: `translateX(${i * 100}%)`, width: `${100 / tabs.length}%` }} />
    {tabs.map((t, j) => <button key={t} onClick={() => setI(j)} className={j === i ? 'on' : ''}>{t}</button>)}</div>;
};
export default TabsSlide;

'use client';

import './SignalPulse.css';

const SignalPulse = ({ color = '#08A66C', rings = 3 }) => {
  return (
    <div className="sigpulse" style={{ '--sig': color }}>
      <div className="sig-dot" />
      {Array.from({ length: rings }, (_, i) => (
        <span key={i} className="sig-ring" style={{ animationDelay: `${i * 0.7}s` }} />
      ))}
    </div>
  );
};

export default SignalPulse;

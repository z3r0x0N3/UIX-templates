'use client';
import { useState } from 'react';
import './PlayMorph.css';
const PlayMorph = () => {
  const [play, setPlay] = useState(false);
  return <button onClick={() => setPlay(!play)} className="playmorph" aria-label={play ? 'pause' : 'play'}>
    <svg viewBox="0 0 24 24"><path className="pm-p" d={play ? 'M7 4h4v16H7zM13 4h4v16h-4z' : 'M7 4l13 8-13 8z'} /></svg></button>;
};
export default PlayMorph;

'use client';
import { useState } from 'react';
import './LikeBurst.css';
const LikeBurst = () => {
  const [liked, setLiked] = useState(false);
  const [burst, setBurst] = useState(0);
  const like = () => { setLiked(true); setBurst(b => b + 1); };
  return <button onClick={like} className={`likeb${liked ? ' on' : ''}`} aria-pressed={liked}>
    <span className="likeb-heart">♥</span>
    {liked && <span className="likeb-minis" key={burst}>
      {Array.from({ length: 7 }, (_, i) => <i key={i} style={{ '--a': `${i * 51}deg` }}>♥</i>)}</span>}
    <span className="likeb-n">{128 + (liked ? 1 : 0)}</span></button>;
};
export default LikeBurst;

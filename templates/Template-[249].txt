'use client';
import { useEffect, useRef } from 'react';
const SparklerWave = () => {
  const ref = useRef(null);
  useEffect(() => {
    const c = ref.current, x = c.getContext('2d');
    let mx = 160, my = 120, sparks = [], raf = 0;
    const onMove = e => { const r = c.getBoundingClientRect();
      mx = (e.clientX - r.left) * (c.width / r.width);
      my = (e.clientY - r.top) * (c.height / r.height); };
    const tip = { x: mx, y: my };
    const tick = () => { x.fillStyle = 'rgba(5,5,10,0.3)'; x.fillRect(0, 0, c.width, c.height);
      tip.x += (mx - tip.x) * 0.2; tip.y += (my - tip.y) * 0.2;
      for (let i = 0; i < 3; i++) sparks.push({ x: tip.x, y: tip.y,
        vx: (Math.random() - 0.5) * 3, vy: (Math.random() - 0.5) * 3 - 0.5, life: 1 });
      sparks = sparks.filter(s => s.life > 0);
      sparks.forEach(s => { s.x += s.vx; s.y += s.vy; s.vy += 0.02; s.life -= 0.03;
        x.fillStyle = `rgba(255,${180 + Math.floor(60 * s.life)},80,${s.life})`;
        x.fillRect(s.x, s.y, 1.6, 1.6); });
      x.strokeStyle = '#888'; x.beginPath(); x.moveTo(tip.x, tip.y); x.lineTo(tip.x - 20, tip.y + 40); x.stroke();
      x.fillStyle = '#fff'; x.beginPath(); x.arc(tip.x, tip.y, 2.4, 0, 6.29); x.fill();
      raf = requestAnimationFrame(tick); };
    raf = requestAnimationFrame(tick);
    c.addEventListener('mousemove', onMove);
    return () => { cancelAnimationFrame(raf); c.removeEventListener('mousemove', onMove); };
  }, []);
  return <canvas ref={ref} width={320} height={180} className="fx-sparkler" />;
};
export default SparklerWave;

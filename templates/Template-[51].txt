'use client';

import { useEffect, useState } from 'react';
import './WordRotator.css';

const WordRotator = ({ lead = 'We build', words = ['interfaces', 'systems', 'futures'], dwell = 2200 }) => {
  const [i, setI] = useState(0);
  const [out, setOut] = useState(false);

  useEffect(() => {
    const iv = setInterval(() => {
      setOut(true);
      setTimeout(() => {
        setI(v => (v + 1) % words.length);
        setOut(false);
      }, 350);
    }, dwell);
    return () => clearInterval(iv);
  }, [words.length, dwell]);

  return (
    <div className="wrot">
      <span className="wrot-lead">{lead}</span>
      <span className={`wrot-word${out ? ' out' : ''}`} key={i}>
        {words[i]}
      </span>
    </div>
  );
};

export default WordRotator;

'use client';

import { useEffect, useRef } from 'react';

const RadarSweep = ({ size = 180, sweepColor = '#00FF41', blips = 5 }) => {
  const ref = useRef(null);

  useEffect(() => {
    const canvas = ref.current;
    const ctx = canvas.getContext('2d');
    const R = size / 2;
    const targets = Array.from({ length: blips }, () => ({
      a: Math.random() * Math.PI * 2,
      r: R * (0.25 + Math.random() * 0.7),
      heat: 0
    }));
    let angle = 0;

    const tick = () => {
      angle += 0.03;
      ctx.fillStyle = 'rgba(4, 12, 4, 0.18)';
      ctx.fillRect(0, 0, size, size);
      ctx.save();
      ctx.translate(R, R);
      // rings
      ctx.strokeStyle = 'rgba(0, 255, 65, 0.35)';
      [0.33, 0.66, 1].forEach(k => {
        ctx.beginPath();
        ctx.arc(0, 0, R * k, 0, Math.PI * 2);
        ctx.stroke();
      });
      // cross
      ctx.beginPath();
      ctx.moveTo(-R, 0); ctx.lineTo(R, 0);
      ctx.moveTo(0, -R); ctx.lineTo(0, R);
      ctx.stroke();
      // sweep
      const grad = ctx.createConicGradient
        ? null
        : null;
      ctx.save();
      ctx.rotate(angle);
      const sg = ctx.createLinearGradient(0, 0, R, 0);
      sg.addColorStop(0, 'rgba(0,255,65,0)');
      sg.addColorStop(1, sweepColor);
      ctx.strokeStyle = sg;
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.moveTo(0, 0); ctx.lineTo(R, 0);
      ctx.stroke();
      ctx.restore();
      // blips flare when sweep passes
      targets.forEach(t => {
        let d = (angle - t.a) % (Math.PI * 2);
        if (d < 0) d += Math.PI * 2;
        t.heat = d < 0.4 ? 1 : Math.max(0, t.heat - 0.02);
        const x = Math.cos(t.a) * t.r;
        const y = Math.sin(t.a) * t.r;
        ctx.fillStyle = `rgba(0, 255, 65, ${0.25 + t.heat * 0.75})`;
        ctx.beginPath();
        ctx.arc(x, y, 2 + t.heat * 3, 0, Math.PI * 2);
        ctx.fill();
      });
      ctx.restore();
      requestAnimationFrame(tick);
    };
    const raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, [size, sweepColor, blips]);

  return <canvas ref={ref} width={size} height={size} className="radar-sweep" />;
};

export default RadarSweep;

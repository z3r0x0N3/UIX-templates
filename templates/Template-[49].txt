'use client';

import { useEffect, useRef } from 'react';

const ParticleOrbit = ({ bodies = 90, rings = [34, 58, 82], coreColor = '#00F0FF' }) => {
  const ref = useRef(null);

  useEffect(() => {
    const canvas = ref.current;
    const ctx = canvas.getContext('2d');
    const cx = canvas.width / 2;
    const cy = canvas.height / 2;
    const ps = Array.from({ length: bodies }, () => {
      const ring = rings[Math.floor(Math.random() * rings.length)];
      return {
        ring,
        a: Math.random() * Math.PI * 2,
        sp: (0.004 + Math.random() * 0.012) * (Math.random() > 0.5 ? 1 : -1),
        s: 0.8 + Math.random() * 1.8
      };
    });

    const tick = () => {
      ctx.fillStyle = 'rgba(5, 8, 14, 0.25)';
      ctx.fillRect(0, 0, canvas.width, canvas.height);
      // core
      const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, 14);
      g.addColorStop(0, '#FFFFFF');
      g.addColorStop(0.4, coreColor);
      g.addColorStop(1, 'rgba(0,0,0,0)');
      ctx.fillStyle = g;
      ctx.beginPath();
      ctx.arc(cx, cy, 14, 0, Math.PI * 2);
      ctx.fill();
      // rings
      ctx.strokeStyle = 'rgba(120,160,200,0.18)';
      rings.forEach(r => {
        ctx.beginPath();
        ctx.arc(cx, cy, r, 0, Math.PI * 2);
        ctx.stroke();
      });
      // bodies
      ps.forEach(p => {
        p.a += p.sp;
        ctx.fillStyle = 'rgba(200,230,255,0.9)';
        ctx.beginPath();
        ctx.arc(cx + Math.cos(p.a) * p.ring, cy + Math.sin(p.a) * p.ring * 0.62, p.s, 0, Math.PI * 2);
        ctx.fill();
      });
      requestAnimationFrame(tick);
    };
    const raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, [bodies, coreColor]);

  return <canvas ref={ref} width={320} height={180} className="particle-orbit" />;
};

export default ParticleOrbit;

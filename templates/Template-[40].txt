'use client';

import './AuroraMesh.css';

const BLOBS = [
  { color: '#00F0FF', x: '10%', y: '20%', size: 220, dur: '11s' },
  { color: '#FF003C', x: '70%', y: '10%', size: 260, dur: '14s' },
  { color: '#8A00FF', x: '40%', y: '70%', size: 240, dur: '9s' },
  { color: '#08A66C', x: '80%', y: '75%', size: 200, dur: '12s' }
];

const AuroraMesh = ({ children, blur = 60 }) => {
  return (
    <div className="aurora-mesh">
      {BLOBS.map((b, i) => (
        <div
          key={i}
          className="aurora-blob"
          style={{
            background: b.color,
            left: b.x,
            top: b.y,
            width: b.size,
            height: b.size,
            filter: `blur(${blur}px)`,
            animationDuration: b.dur
          }}
        />
      ))}
      <div className="aurora-content">{children}</div>
    </div>
  );
};

export default AuroraMesh;

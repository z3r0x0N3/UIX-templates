'use client';
import { useEffect, useRef } from 'react';
const Oscilloscope = () => {
  const ref = useRef(null);
  useEffect(() => {
    const c = ref.current, x = c.getContext('2d');
    let raf = 0, t = 0;
    const tick = () => { t += 0.09;
      x.fillStyle = '#031007'; x.fillRect(0, 0, c.width, c.height);
      x.strokeStyle = 'rgba(0,255,65,0.18)';
      for (let gx = 0; gx <= c.width; gx += 32) { x.beginPath(); x.moveTo(gx, 0); x.lineTo(gx, c.height); x.stroke(); }
      for (let gy = 0; gy <= c.height; gy += 30) { x.beginPath(); x.moveTo(0, gy); x.lineTo(c.width, gy); x.stroke(); }
      x.strokeStyle = '#00FF41'; x.lineWidth = 2; x.beginPath();
      for (let px = 0; px <= c.width; px += 3) {
        const y = c.height / 2 + Math.sin(px * 0.06 + t) * 34 * Math.sin(t * 0.4) + (Math.random() - 0.5) * 5;
        px ? x.lineTo(px, y) : x.moveTo(px, y); }
      x.stroke(); raf = requestAnimationFrame(tick); };
    raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, []);
  return <canvas ref={ref} width={320} height={180} className="fx-scope" />;
};
export default Oscilloscope;

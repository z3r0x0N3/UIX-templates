<!-- AcidSquares — correct working implementation (vanilla Canvas, no dependencies).
     Props: color1/2/3, speed, waveDepth, zoom, density (block count),
     glow, exposure, spread, colorShift, contrast, brightness, opacity,
     mouseInteraction, mouseStrength, mouseRadius (fraction of field),
     blur, grain, grainIntensity.
     Renders domain-warped acid blocks (smoothing OFF = squares) at low
     resolution, upscaled; mouse warps the field; grain overlay optional.
     Usage:
       <div data-acidsquares style="width:100%;height:600px"></div>
       <script> /* paste everything below */ </script> -->
<script>
  function AcidSquares(canvas, o) {
    o = Object.assign({ color1: '#5227FF', color2: '#A855F7', color3: '#FFFFFF',
      speed: 0.7, waveDepth: 1, zoom: 1.3, density: 10, glow: 1, exposure: 2700,
      spread: 0.3, colorShift: 0, contrast: 1, brightness: 1, opacity: 1,
      mouseInteraction: true, mouseStrength: 0.1, mouseRadius: 0.35, blur: 0,
      grain: true, grainIntensity: 0.05 }, o || {});
    const hex = h => { const m = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(h);
      return m ? [parseInt(m[1], 16), parseInt(m[2], 16), parseInt(m[3], 16)] : [255, 255, 255]; };
    const c1 = hex(o.color1), c2 = hex(o.color2), c3 = hex(o.color3);
    const x = canvas.getContext('2d');
    const LW = Math.max(24, Math.round(o.density * 12));
    const LH = Math.max(12, Math.round(o.density * 6));
    const buf = document.createElement('canvas'); buf.width = LW; buf.height = LH;
    const bx = buf.getContext('2d');
    const img = bx.createImageData(LW, LH);
    const smooth = (a, b, t) => { t = Math.max(0, Math.min(1, (t - a) / (b - a))); return t * t * (3 - 2 * t); };
    let mx = -999, my = -999;
    if (o.mouseInteraction) {
      canvas.addEventListener('mousemove', e => {
        const r = canvas.getBoundingClientRect();
        mx = (e.clientX - r.left) / r.width * LW;
        my = (e.clientY - r.top) / r.height * LH;
      });
      canvas.addEventListener('mouseleave', () => { mx = my = -999; });
    }
    const gain = o.exposure / 2700 * o.brightness;
    let t = Math.random() * 10, last = performance.now();
    function frame(now) {
      t += (now - last) / 1000 * o.speed; last = now;
      const d = img.data, R = o.mouseRadius * Math.max(LW, LH);
      for (let j = 0; j < LH; j++) for (let i = 0; i < LW; i++) {
        let nx = i / LW * o.zoom * 4, ny = j / LH * o.zoom * 4;
        if (mx > -100) {
          const dx = i - mx, dy = j - my, dist = Math.sqrt(dx * dx + dy * dy);
          if (dist < R && dist > 0.001) {
            const f = (1 - dist / R) * o.mouseStrength * 8;
            nx += dx / dist * f; ny += dy / dist * f;
          }
        }
        const w1 = Math.sin(nx * 1.7 + t * 1.4 + Math.sin(ny * 2.2 - t) * o.waveDepth);
        const w2 = Math.cos(ny * 2.0 - t * 1.1 + Math.sin(nx * 1.4 + t * 0.7) * o.waveDepth);
        let v = 0.5 + 0.5 * Math.sin(w1 + w2 + t * 0.3);
        v = (((v + o.colorShift) % 1) + 1) % 1;
        const a = smooth(0.5 - o.spread, 0.5 + o.spread * 0.3, v);
        let r = c1[0] + (c2[0] - c1[0]) * a,
            g = c1[1] + (c2[1] - c1[1]) * a,
            b = c1[2] + (c2[2] - c1[2]) * a;
        const bb = smooth(0.5, 0.5 + o.spread, v);
        r += (c3[0] - r) * bb; g += (c3[1] - g) * bb; b += (c3[2] - b) * bb;
        const gl = Math.pow(v, 3) * o.glow * 90;
        r += gl; g += gl; b += gl;
        r *= gain; g *= gain; b *= gain;
        r = (r - 128) * o.contrast + 128;
        g = (g - 128) * o.contrast + 128;
        b = (b - 128) * o.contrast + 128;
        if (o.grain) {
          const gr = (Math.random() - 0.5) * o.grainIntensity * 512;
          r += gr; g += gr; b += gr;
        }
        const k = (j * LW + i) * 4;
        d[k] = r > 255 ? 255 : r < 0 ? 0 : r;
        d[k + 1] = g > 255 ? 255 : g < 0 ? 0 : g;
        d[k + 2] = b > 255 ? 255 : b < 0 ? 0 : b;
        d[k + 3] = 255;
      }
      bx.putImageData(img, 0, 0);
      x.save();
      x.globalAlpha = o.opacity;
      x.imageSmoothingEnabled = false; // squares stay square
      if (o.blur > 0) x.filter = 'blur(' + o.blur + 'px)';
      x.clearRect(0, 0, canvas.width, canvas.height);
      x.drawImage(buf, 0, 0, canvas.width, canvas.height);
      x.restore();
      requestAnimationFrame(frame);
    }
    requestAnimationFrame(frame);
  }

  // Defaults match the snippet's props exactly.
  document.querySelectorAll('[data-acidsquares]').forEach(el => {
    const canvas = document.createElement('canvas');
    canvas.width = 960; canvas.height = 600;
    canvas.style.width = '100%'; canvas.style.height = '100%';
    el.appendChild(canvas);
    AcidSquares(canvas, {
      color1: '#5227FF', color2: '#A855F7', color3: '#FFFFFF',
      speed: 0.7, waveDepth: 1, zoom: 1.3, density: 10, glow: 1, exposure: 2700,
      spread: 0.3, colorShift: 0, contrast: 1, brightness: 1, opacity: 1,
      mouseInteraction: true, mouseStrength: 0.1, mouseRadius: 0.35,
      blur: 0, grain: true, grainIntensity: 0.05
    });
  });
</script>

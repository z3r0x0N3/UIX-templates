'use client';
import { useState } from 'react';
import './DominoTopple.css';
const DominoTopple = ({ n = 7 }) => {
  const [k, setK] = useState(0);
  const topple = () => setK(x => x + 1);
  return <button onClick={topple} className="dominoes" key={k} aria-label="topple dominoes">
    {Array.from({ length: n }, (_, i) => (
      <span key={i} className="domino" style={{ animationDelay: `${0.1 + i * 0.22}s` }} />))}</button>;
};
export default DominoTopple;

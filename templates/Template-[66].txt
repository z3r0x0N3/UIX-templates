'use client';
import { useEffect, useState } from 'react';
import './BlurReveal.css';
const BlurReveal = ({ text = 'Focus travels word by word', period = 2600 }) => {
  const [k, setK] = useState(0);
  useEffect(() => { const iv = setInterval(() => setK(v => v + 1), period); return () => clearInterval(iv); }, [period]);
  const words = text.split(' ');
  return <div className="breveal" key={k}>{words.map((w, i) => (
    <span key={i} className="breveal-w" style={{ animationDelay: `${i * 0.22}s` }}>{w}</span>))}</div>;
};
export default BlurReveal;
